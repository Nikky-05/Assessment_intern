import pandas as pd
import numpy as np

# ------------------ Load & Clean ------------------
file_path = "cleaned_sales_data.csv"
df = pd.read_csv(file_path)
df = df.drop_duplicates()
df['PurchaseDate'] = pd.to_datetime(df['PurchaseDate'], errors='coerce')
df['Quantity'] = pd.to_numeric(df['Quantity'], errors='coerce').fillna(0).astype(int)
df['Price'] = pd.to_numeric(df['Price'], errors='coerce').fillna(0.0)
df['FeedbackScore'] = pd.to_numeric(df['FeedbackScore'], errors='coerce')
df['FeedbackScore'] = df['FeedbackScore'].fillna(df['FeedbackScore'].median())
df['Returned'] = df['Returned'].map(
    {True: True, 'True': True, 'true': True, 'Yes': True, 'Y': True,
     False: False, 'False': False, 'false': False, 'No': False, 'N': False}
).astype('boolean')
df['Sales'] = df['Quantity'] * df['Price']
df['Year'] = df['PurchaseDate'].dt.year
df['Month'] = df['PurchaseDate'].dt.month
df['MonthName'] = df['PurchaseDate'].dt.strftime('%Y-%m')
df['Weekday'] = df['PurchaseDate'].dt.day_name()
df['Quarter'] = df['PurchaseDate'].dt.quarter

# Get unique countries for dropdown
countries = ['All Countries'] + sorted(df['Country'].unique().tolist())

    return fig

# ------------------ Callbacks ------------------
@callback(
    [Output('total-sales-kpi', 'children'),
     Output('unique-customers-kpi', 'children'),
     Output('avg-order-value-kpi', 'children'),
     Output('return-rate-kpi', 'children'),
     Output('avg-rating-kpi', 'children'),
     Output('sales-trend-chart', 'figure'),
     Output('category-distribution-chart', 'figure'),
     Output('top-products-chart', 'figure'),
     Output('weekday-sales-chart', 'figure'),
     Output('satisfaction-chart', 'figure'),
     Output('geographic-performance-chart', 'figure'),
     Output('return-analysis-chart', 'figure')],
    [Input('country-dropdown', 'value')]
)
def update_dashboard(selected_country):
    # Filter data based on country selection
    if selected_country == 'All Countries':
        filtered_df = df
        title_suffix = "All Countries"
    else:
        filtered_df = df[df['Country'] == selected_country]
        title_suffix = selected_country
    
    # Calculate KPIs
    total_sales = filtered_df['Sales'].sum()
    unique_customers = filtered_df['CustomerID'].nunique()
    avg_order_value = filtered_df.groupby('CustomerID')['Sales'].sum().mean() if unique_customers > 0 else 0
    return_rate = filtered_df['Returned'].mean() if len(filtered_df) > 0 else 0
    avg_rating = filtered_df['FeedbackScore'].mean() if len(filtered_df) > 0 else 0
    
    # Format KPI values
    total_sales_formatted = f"${total_sales:,.0f}" if total_sales > 0 else "$0"
    unique_customers_formatted = f"{unique_customers:,}" if unique_customers > 0 else "0"
    avg_order_value_formatted = f"${avg_order_value:,.0f}" if avg_order_value > 0 else "$0"
    return_rate_formatted = f"{return_rate*100:.1f}%" if len(filtered_df) > 0 else "0.0%"
    avg_rating_formatted = f"{avg_rating:.1f}/5" if avg_rating > 0 else "0.0/5"
    
    # Chart 1: Sales Trend by Month (Area Chart)
    monthly_data = filtered_df.groupby('MonthName').agg({
        'Sales': 'sum',
        'Quantity': 'sum',
        'CustomerID': 'nunique'
    }).reset_index()
    
    if not monthly_data.empty:
        monthly_data['MonthName'] = pd.to_datetime(monthly_data['MonthName'])
        monthly_data = monthly_data.sort_values('MonthName')
        fig_trend = px.area(
            monthly_data, 
            x='MonthName', 
            y='Sales',
            title=f"📈 Monthly Sales Trend - {title_suffix}",
            labels={'Sales': 'Sales Amount', 'MonthName': 'Month'}
        )
        fig_trend.update_traces(fill='tozeroy', line=dict(width=3))
    else:
        fig_trend = px.area(title=f"No Sales Data Available - {title_suffix}")
    
    # Chart 2: Category Distribution (Donut Chart)
    category_data = filtered_df.groupby('Category').agg({
        'Sales': 'sum',
        'Quantity': 'sum'
    }).reset_index()
    
    if not category_data.empty:
        fig_category = px.pie(
            category_data, 
            values='Sales', 
            names='Category',
            title=f"📊 Sales by Category - {title_suffix}",
            hole=0.4
        )
        fig_category.update_traces(textposition='inside', textinfo='percent+label')
    else:
        fig_category = px.pie(title=f"No Category Data - {title_suffix}")
    
    # Chart 3: Top Products Performance (Horizontal Bar Chart)
    top_products = filtered_df.groupby('Product').agg({
        'Sales': 'sum',
        'Quantity': 'sum',
        'CustomerID': 'nunique'
    }).nlargest(8, 'Sales').reset_index()
    
    if not top_products.empty:
        fig_products = px.bar(
            top_products, 
            y='Product', 
            x='Sales',
            title=f"🏆 Top Performing Products - {title_suffix}",
            orientation='h',
            color='Sales',
            labels={'Sales': 'Total Sales', 'Product': ''}
        )
        fig_products.update_layout(yaxis={'categoryorder':'total ascending'})
    else:
        fig_products = px.bar(title=f"No Product Data - {title_suffix}")
    
    # Chart 4: Sales by Weekday (Bar Chart)
    weekday_order = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday']
    weekday_data = filtered_df.groupby('Weekday').agg({
        'Sales': 'sum',
        'Quantity': 'sum'
    }).reset_index()
    weekday_data['Weekday'] = pd.Categorical(weekday_data['Weekday'], categories=weekday_order, ordered=True)
    weekday_data = weekday_data.sort_values('Weekday')
    
    if not weekday_data.empty:
        fig_weekday = px.bar(
            weekday_data, 
            x='Weekday', 
            y='Sales',
            title=f"📅 Sales by Weekday - {title_suffix}",
            color='Sales',
            labels={'Sales': 'Total Sales', 'Weekday': 'Day of Week'}
        )
    else:
        fig_weekday = px.bar(title=f"No Weekday Data - {title_suffix}")
    
    # Chart 5: Customer Satisfaction (Gauge Chart)
    rating_distribution = filtered_df['FeedbackScore'].value_counts().sort_index()
    
    if not rating_distribution.empty:
        fig_satisfaction = go.Figure(go.Indicator(
            mode = "gauge+number+delta",
            value = avg_rating,
            domain = {'x': [0, 1], 'y': [0, 1]},
            title = {'text': f"⭐ Customer Satisfaction - {title_suffix}"},
            gauge = {
                'axis': {'range': [None, 5], 'tickwidth': 1, 'tickcolor': "white"},
                'bar': {'color': "gold"},
                'steps': [
                    {'range': [0, 2], 'color': 'red'},
                    {'range': [2, 4], 'color': 'yellow'},
                    {'range': [4, 5], 'color': 'green'}
                ]
            }
        ))
    else:
        fig_satisfaction = go.Figure()
        fig_satisfaction.add_annotation(text=f"No Rating Data - {title_suffix}", x=0.5, y=0.5)
    
    # Chart 6: Geographic Performance (Top Cities)
    city_data = filtered_df.groupby('City').agg({
        'Sales': 'sum',
        'CustomerID': 'nunique'
    }).nlargest(10, 'Sales').reset_index()
    
    if not city_data.empty:
        fig_geographic = px.bar(
            city_data, 
            x='Sales', 
            y='City',
            title=f"🌍 Top Cities by Sales - {title_suffix}",
            orientation='h',
            color='CustomerID',
            labels={'Sales': 'Total Sales', 'City': '', 'CustomerID': 'Customers'}
        )
        fig_geographic.update_layout(yaxis={'categoryorder':'total ascending'})
    else:
        fig_geographic = px.bar(title=f"No Geographic Data - {title_suffix}")
    
    # Chart 7: Return Analysis
    return_data = filtered_df.groupby('Product').agg({
        'Returned': 'sum',
        'Quantity': 'sum'
    }).reset_index()
    return_data['Return Rate'] = (return_data['Returned'] / return_data['Quantity'] * 100).fillna(0)
    
    if not return_data.empty:
        fig_return = px.scatter(
            return_data,
            x='Quantity',
            y='Return Rate',
            size='Returned',
            color='Product',
            title=f"🔄 Return Analysis by Product - {title_suffix}",
            labels={'Quantity': 'Units Sold', 'Return Rate': 'Return Rate (%)', 'Returned': 'Returns Count'}
        )
    else:
        fig_return = px.scatter(title=f"No Return Data - {title_suffix}")
    
    # Apply transparent theme to all charts
    plotly_charts = [fig_trend, fig_category, fig_products, fig_weekday, fig_geographic, fig_return]
    for chart in plotly_charts:
        transparent_chart(chart)
    
    # Apply styling to gauge chart
    transparent_chart(fig_satisfaction)
    
    return (
        total_sales_formatted,
        unique_customers_formatted,
        avg_order_value_formatted,
        return_rate_formatted,
        avg_rating_formatted,
        fig_trend,
        fig_category,
        fig_products,
        fig_weekday,
        fig_satisfaction,
        fig_geographic,
        fig_return
    )

if __name__ == "__main__":
    app.run(debug=True)
